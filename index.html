HARDWARE PRACTICE

บทเรียนและโค้ดตัวอย่าง

1. LED กระพริบ

ต่อ GPIO 25 ผ่านตัวต้านทาน 220–330Ω ไปยังขาบวก LED และต่อขาลบลง GND

const int LED = 25;

void setup() {
  pinMode(LED, OUTPUT);
}

void loop() {
  digitalWrite(LED, HIGH);
  delay(1000);
  digitalWrite(LED, LOW);
  delay(1000);
}

โจทย์ดัดแปลง: ให้ LED ติด 200 ms และดับ 800 ms

2. RGB LED เปลี่ยนสี

ตัวอย่างนี้ใช้ RGB แบบ Common Cathode โดยแต่ละช่องสีต้องมีตัวต้านทานของตัวเอง

  • R → GPIO 25
  • G → GPIO 26
  • B → GPIO 27
  • ขาร่วม → GND
const int R = 25, G = 26, B = 27;

void setup() {
  pinMode(R, OUTPUT);
  pinMode(G, OUTPUT);
  pinMode(B, OUTPUT);
}

void loop() {
  // สีแดง
  digitalWrite(R, HIGH);
  digitalWrite(G, LOW);
  digitalWrite(B, LOW);
  delay(1000);

  // สีเขียว
  digitalWrite(R, LOW);
  digitalWrite(G, HIGH);
  digitalWrite(B, LOW);
  delay(1000);

  // สีน้ำเงิน
  digitalWrite(R, LOW);
  digitalWrite(G, LOW);
  digitalWrite(B, HIGH);
  delay(1000);

  // สีเหลือง = แดง + เขียว
  digitalWrite(R, HIGH);
  digitalWrite(G, HIGH);
  digitalWrite(B, LOW);
  delay(1000);
}

Common Anode ต้องกลับตรรกะ HIGH/LOW และต่อขาร่วมตามชนิดอุปกรณ์

3. Ultrasonic HC-SR04

Trig สั่งส่งคลื่นเสียง, Echo ให้สัญญาณสำหรับจับเวลา

const int TRIG = 5;
const int ECHO = 18;
const int LED = 25;

void setup() {
  Serial.begin(115200);
  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);
  pinMode(LED, OUTPUT);
}

void loop() {
  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG, LOW);

  long duration = pulseIn(ECHO, HIGH, 30000);

  if (duration == 0) {
    Serial.println("No echo");
    digitalWrite(LED, LOW);
  } else {
    float cm = duration * 0.0343 / 2;
    Serial.print("Distance: ");
    Serial.print(cm);
    Serial.println(" cm");

    digitalWrite(LED, cm < 10 ? HIGH : LOW);
  }
  delay(200);
}

ระยะทาง (cm) = เวลา Echo (µs) × 0.0343 ÷ 2

4. LCD 1602 I2C

ตัวอย่างใช้ Address 0x27 และ ESP32 SDA=21, SCL=22 (ตรวจสอบ Address/ขาจริงของบอร์ด)

คำสั่งหน้าที่
lcd.init()เริ่มต้นจอ
lcd.backlight()เปิดไฟพื้นหลัง
lcd.noBacklight()ปิดไฟพื้นหลัง
lcd.clear()ล้างข้อความ
lcd.setCursor(col,row)กำหนดตำแหน่ง (เริ่มนับจาก 0)
lcd.print()แสดงข้อความ/ตัวเลข
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

void setup() {
  Wire.begin(21, 22);
  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("Hello Pae!");
  lcd.setCursor(0, 1);
  lcd.print("IOT SYSTEM");
}

void loop() {}

CHALLENGE MODE

15 นาที

A. ไฟจราจรจำลอง

เขียว 3 วินาที → เหลือง 1 วินาที → แดง 3 วินาที แล้ววนซ้ำ

แนวคิดเฉลย

กำหนด GPIO แยก 3 ขา ตั้ง OUTPUT ทั้งหมด แล้วเปิดทีละสีตามลำดับ ปิดสีอื่นทุกครั้งก่อนเปลี่ยนสี

20 นาที

B. RGB ตามระยะ

น้อยกว่า 10 cm = แดง, 10–30 cm = เหลือง, มากกว่า 30 cm = เขียว

แนวคิดเฉลย

อ่านระยะจาก HC-SR04 แล้วใช้ if / else if / else กำหนด R,G,B: แดง (1,0,0), เหลือง (1,1,0), เขียว (0,1,0) สำหรับ Common Cathode

30 นาที

C. ระบบ DANGER / SAFE

ระยะน้อยกว่า 10 cm ให้ LED ติดและ Serial แสดง DANGER มิฉะนั้นแสดง SAFE

แนวคิดเฉลย

ตรวจ duration==0 ก่อนคำนวณ จากนั้นคำนวณ cm แล้วใช้ if (cm < 10) เปิด LED และพิมพ์ DANGER; else ปิด LED และพิมพ์ SAFE
